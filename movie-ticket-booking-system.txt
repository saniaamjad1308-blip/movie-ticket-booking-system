
CREATE TABLE Movies (
    movie_id NUMBER PRIMARY KEY,
    title VARCHAR2(100) NOT NULL,
    genre VARCHAR2(50),
    duration NUMBER
);

CREATE TABLE Customers (
    customer_id NUMBER PRIMARY KEY,
    customer_name VARCHAR2(100) NOT NULL,
    email VARCHAR2(100) UNIQUE,
    phone VARCHAR2(15)
);

CREATE TABLE Shows (
    show_id NUMBER PRIMARY KEY,
    movie_id NUMBER NOT NULL,
    show_date DATE NOT NULL,
    show_time VARCHAR2(10) NOT NULL,
    ticket_price NUMBER(8,2) CHECK (ticket_price > 0),
    available_seats NUMBER CHECK (available_seats >= 0),
    FOREIGN KEY (movie_id) REFERENCES Movies(movie_id)
);

CREATE TABLE Bookings (
    booking_id NUMBER PRIMARY KEY,
    customer_id NUMBER NOT NULL,
    show_id NUMBER NOT NULL,
    tickets_booked NUMBER CHECK (tickets_booked > 0),
    booking_date DATE NOT NULL,
    FOREIGN KEY (customer_id) REFERENCES Customers(customer_id),
    FOREIGN KEY (show_id) REFERENCES Shows(show_id)
);

-- 1. Display all movies
SELECT * FROM Movies;

-- 2. Display all customers
SELECT * FROM Customers;

-- 3. Display shows with movie titles
SELECT S.show_id, M.title, S.show_date,
       S.show_time, S.ticket_price
FROM Shows S
JOIN Movies M ON S.movie_id = M.movie_id;

-- 4. Display booking details and calculate amount
SELECT B.booking_id, C.customer_name, M.title,
       B.tickets_booked, S.ticket_price,
       B.tickets_booked * S.ticket_price AS total_amount
FROM Bookings B
JOIN Customers C ON B.customer_id = C.customer_id
JOIN Shows S ON B.show_id = S.show_id
JOIN Movies M ON S.movie_id = M.movie_id;

-- 5. Calculate total revenue
SELECT SUM(B.tickets_booked * S.ticket_price) AS revenue
FROM Bookings B
JOIN Shows S ON B.show_id = S.show_id;

-- 6. Count bookings for each movie
SELECT M.title, COUNT(B.booking_id) AS total_bookings
FROM Movies M
JOIN Shows S ON M.movie_id = S.movie_id
JOIN Bookings B ON S.show_id = B.show_id
GROUP BY M.title;

-- 7. Find bookings above Rs. 500
SELECT B.booking_id,
       B.tickets_booked * S.ticket_price AS amount
FROM Bookings B
JOIN Shows S ON B.show_id = S.show_id
WHERE B.tickets_booked * S.ticket_price > 500;

-- 8. Movie-wise revenue
SELECT M.title,
       SUM(B.tickets_booked * S.ticket_price) AS revenue
FROM Movies M
JOIN Shows S ON M.movie_id = S.movie_id
JOIN Bookings B ON S.show_id = B.show_id
GROUP BY M.title
ORDER BY revenue DESC;

-- 9. Movies without bookings
SELECT M.title
FROM Movies M
WHERE NOT EXISTS (
    SELECT 1
    FROM Shows S
    JOIN Bookings B ON S.show_id = B.show_id
    WHERE S.movie_id = M.movie_id
);

-- 10. Total bookings
SELECT COUNT(*) AS total_bookings FROM Bookings;

SET ticket_price = 210
